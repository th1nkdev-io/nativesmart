# @nativesmart/react-native

Expo-friendly React Native components, hooks and themes for Nativesmart mobile products.

## Usage

```tsx
import { Button, NativesmartProvider } from "@nativesmart/react-native";

export function App() {
  return (
    <NativesmartProvider>
      <Button label="Continue" onPress={() => {}} />
    </NativesmartProvider>
  );
}
```
